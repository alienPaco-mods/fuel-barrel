    model FuelBarrelGround
    {
        mesh = FuelBarrel,
        texture = FuelBarrelModel,
        scale = 0.54,
    }

    model CrudeFuelBarrelGround
    {
        mesh = CrudeFuelBarrel,
        texture = CrudeFuelBarrelModel,
        scale = 0.54,

    item FuelBarrel
    {
        DisplayCategory = Tool,
        ItemType = base:normal,

        component FluidContainer

    {

    craftRecipe MakeFuelBarrel
    {
        OnCreate = FuelBarrel.onCreateBarrel,
        timedAction = Welding,
        time = 250,
        SkillRequired = MetalWelding:6,
        Tags = AnySurfaceCraft;Welding,
        category = Metalworking,
        xpAward = MetalWelding:60,
        inputs
        {
            item 1 tags[base:weldingmask] mode:keep,
            item 4 [Base.BlowTorch],
            item 4 [Base.SheetMetal],
            item 4 [Base.WeldingRods],
        }
        outputs
        {
            item 1 FuelBarrel.FuelBarrel,
        }
    }

            item 2 [Base.SheetMetal],
